# Optional dependency for the P3.1 explainability layer (config: explain.enabled).
#
# Kept out of the base requirements.txt / requirements.lock on purpose: the
# feature is DEFAULT OFF and advisory-only, so a deployment that never enables it
# should not ship an LLM SDK. When explain.enabled is turned on (and
# ANTHROPIC_API_KEY is set), install this on top of the base requirements:
#
#   pip install -r requirements.lock -r requirements-explain.txt
#
# The /api/explain endpoint degrades gracefully (returns {"available": false})
# when this is absent, so omitting it never breaks a deployment.
# Default provider (explain.provider = "anthropic"):
anthropic==0.121.0

# OpenAI-compatible provider instead (explain.provider = "openai", i.e. OpenAI,
# NVIDIA Nemotron, OpenRouter, a local vLLM server, …): install the OpenAI SDK.
# Only one of these is needed — whichever matches explain.provider.
openai==2.53.0
